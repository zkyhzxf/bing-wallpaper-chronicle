### 复核片段（可复跑）

```python
import json, glob, datetime
dates = set()
for f in glob.glob('archive/*.json'):
    d = json.load(open(f, encoding='utf-8'))
    if isinstance(d, list):
        for r in d:
            if isinstance(r, dict) and r.get('d'): dates.add(str(r['d']))
ds = sorted(dates)
a = datetime.date(int(ds[0][:4]), int(ds[0][4:6]), int(ds[0][6:8]))
b = datetime.date(int(ds[-1][:4]), int(ds[-1][4:6]), int(ds[-1][6:8]))
print(len(dates), ds[0], ds[-1], (b-a).days+1, (b-a).days+1-len(dates))
# 实测输出：6339 20090713 20261122 6342 3
```
